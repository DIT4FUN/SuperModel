### 创建对应等级AGV代码示例

```python
from src.sensors.tactile import get_tactile_spec, TactileArray
from src.sensors.force import get_force_spec, SixAxisForceTorque
from src.sensors.imu import get_imu_spec, IMU
from src.control.agv import AGVSpec, AGVGrade, AGVMotionController

# 获取M级规格
tactile_spec = get_tactile_spec('M')
force_spec = get_force_spec('M')
imu_spec = get_imu_spec('M')
agv_spec = AGVSpec.from_grade(AGVGrade.M)

# 创建传感器
tactile = TactileArray(
    rows=16, cols=16,
    sample_rate=tactile_spec['sample_rate']
)
force = SixAxisForceTorque(
    can_id=0x01,
    gravity_compensation=True
)
imu = IMU(
    sample_rate=imu_spec['sample_rate']
)

# 创建AGV运动控制器
agv = AGVMotionController(agv_spec)
```
